# requirements.txt: pemasangan di laptop (Windows dan lintas platform)
#
# Berkas ini untuk pengembangan lokal. Yang dipakai container produksi adalah
# requirements.lock, yang memuat seluruh dependensi turunan beserta hash dan
# dipasang dengan --require-hashes.
#
# Alur pembaruan versi:
#   1. sunting requirements.in atau requirements-dev.in
#   2. pip-compile --generate-hashes --strip-extras --allow-unsafe requirements.in
#   3. pip-compile --generate-hashes --strip-extras --allow-unsafe requirements-dev.in
#   4. sesuaikan berkas ini agar versinya tetap sama
#
# Berkas ini sengaja tidak dihasilkan pip-compile karena memuat penanda
# sys_platform yang perlu dibaca manusia, sementara lockfile produksi hanya
# menyasar Linux.

# Kerangka aplikasi
Flask==3.0.3

# Server
# Penanda sys_platform mencegah pip memasang gunicorn di Windows: paketnya
# terpasang tetapi tidak dapat dijalankan karena bergantung pada modul fcntl
# yang hanya ada di sistem mirip Unix.
waitress==3.0.2                              # Windows dan lintas platform
gunicorn==23.0.0; sys_platform != "win32"    # Linux, dipakai container

# Pengujian dan pemeriksaan
pytest==8.3.3
pytest-randomly==3.15.0
coverage==7.6.1

# Pemeriksaan keamanan. Dijalankan CI pada tiap push, dapat juga dijalankan
# lokal sebelum mendorong perubahan:
#   pip-audit -r requirements.lock     periksa CVE pada dependensi
#   bandit -r . -x ./tests             periksa pola berisiko pada kode
pip-audit==2.9.0
bandit==1.8.0
